## 1.发放植物

运行起来后，在服务端控制台执行

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempGiveMusicIcePeashooter")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempGiveMusicIcePeashooter"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ServerScriptService = game:GetService("ServerScriptService")

-- 植物名字和数量
local PlantName = "PumpkinShield"
local Amount = 6

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)

local player = Players:FindFirstChild("AXrain10")
if not player then
	player = Players.PlayerAdded:Wait()
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end
	task.wait(0.1)
end

if not data then
	warn("[TempGive] Data still not loaded for " .. tostring(player and player.Name))
	script:Destroy()
	return
end
-- 发植物
PlayerDataManager.GivePlant(player, PlantName, Amount, 1, true)
-- 发种子
PlayerDataManager.GiveSeed(player, PlantName, Amount, true)

-- 打印信息
print(string.format("[TempGive] 成功发放！数量: %d, 植物: %s, 玩家: %s", Amount, PlantName, player.Name))

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```

## 2.通行证经验

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempAddPassExp")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempAddPassExp"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ServerScriptService = game:GetService("ServerScriptService")

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)

local TARGET_PLAYER_NAME = "AXrain10"
local EXP_AMOUNT = 150
local FORCE_SAVE = false

local player = Players:FindFirstChild(TARGET_PLAYER_NAME)
if not player then
	player = Players.PlayerAdded:Wait()
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end
	task.wait(0.1)
end

if not data then
	warn("[TempPassExp] Data still not loaded for " .. tostring(player and player.Name))
	script:Destroy()
	return
end

local passData = PlayerDataManager.AddPassExp(player, EXP_AMOUNT)
if not passData then
	warn("[TempPassExp] AddPassExp failed for " .. tostring(player.Name))
	script:Destroy()
	return
end

print(("[TempPassExp] Added %d Pass EXP to %s"):format(EXP_AMOUNT, player.Name))
print(("[TempPassExp] Level=%s Exp=%s RepeatAvailable=%s PremiumOwned=%s"):format(
	tostring(passData.Level),
	tostring(passData.Exp),
	tostring(passData.RepeatRewardAvailable),
	tostring(passData.PremiumOwned)
))

if FORCE_SAVE then
	local ok = PlayerDataManager.ForceSave(player)
	print("[TempPassExp] ForceSave =", ok)
end

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```

## 3.已通关层数

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempSetPlayerMaxLevel")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempSetPlayerMaxLevel"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerScriptService = game:GetService("ServerScriptService")

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)
local RankingSeasonConfig = require(ReplicatedStorage.Shared.RankingSeasonConfig)

local TARGET_PLAYER_NAME = "AXrain10"
local TARGET_MAX_SELECTABLE_LEVEL = 50
local FORCE_SAVE = false

local player = Players:FindFirstChild(TARGET_PLAYER_NAME)
if not player then
	player = Players.PlayerAdded:Wait()
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end
	task.wait(0.1)
end

if not data then
	warn("[TempMaxLevel] Data still not loaded for " .. tostring(player and player.Name))
	script:Destroy()
	return
end

local targetLevel = math.max(1, math.floor(tonumber(TARGET_MAX_SELECTABLE_LEVEL) or 1))
local seasonId = RankingSeasonConfig.GetSeasonId()

if type(data.RankingSeasonUnlockedLevels) ~= "table" then
	data.RankingSeasonUnlockedLevels = {}
end

local oldSeasonLevel = tonumber(data.RankingSeasonUnlockedLevels[seasonId]) or 1
local oldUnlockedLevel = tonumber(data.UnlockedLevel) or 1

data.RankingSeasonUnlockedLevels[seasonId] = targetLevel
data.UnlockedLevel = targetLevel

local updateInventoryRE = ReplicatedStorage:FindFirstChild("UpdateInventoryRE")
if updateInventoryRE then
	updateInventoryRE:FireClient(player, data)
end

local character = player.Character
local head = character and character:FindFirstChild("Head")
local titleGui = head and head:FindFirstChild("PlayerTitleGui")
local titleLabel = titleGui and titleGui:FindFirstChildWhichIsA("TextLabel", true)
if titleLabel then
	titleLabel.Text = "MAX  " .. tostring(math.max(0, targetLevel - 1))
end

print(("[TempMaxLevel] %s season=%s max selectable level: %s -> %s"):format(
	player.Name,
	tostring(seasonId),
	tostring(oldSeasonLevel),
	tostring(targetLevel)
))
print(("[TempMaxLevel] UnlockedLevel: %s -> %s"):format(
	tostring(oldUnlockedLevel),
	tostring(data.UnlockedLevel)
))

if FORCE_SAVE then
	local ok = PlayerDataManager.ForceSave(player)
	print("[TempMaxLevel] ForceSave =", ok)
end

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```

## 4.增加金币

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempGiveSunCoins")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempGiveSunCoins"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ServerScriptService = game:GetService("ServerScriptService")

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)

local TARGET_PLAYER_NAME = "AXrain10"
local SUN_AMOUNT = 100000000000
local FORCE_SAVE = false

local player = Players:FindFirstChild(TARGET_PLAYER_NAME)
if not player then
	player = Players.PlayerAdded:Wait()
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end

end

	warn("[TempGiveSun] Data still not loaded for " .. tostring(player and player.Name))
	script:Destroy()
	return
end

local beforeCoins = tonumber(data.Coins) or 0

local afterData = PlayerDataManager.GetPlayerData(player)

print(("[TempGiveSun] Added %d Sun to %s"):format(SUN_AMOUNT, player.Name))
print(("[TempGiveSun] Coins: %s -> %s"):format(tostring(beforeCoins), tostring(afterCoins)))

if FORCE_SAVE then
	local ok = PlayerDataManager.ForceSave(player)
	print("[TempGiveSun] ForceSave =", ok)
end

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```
