## 1.发放植物

运行起来后，在服务端控制台执行

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempGiveMusicIcePeashooter")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempGiveMusicIcePeashooter"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ServerScriptService = game:GetService("ServerScriptService")

-- 植物名字和数量
local PlantName = "CarrotCannon"
local Amount = 6

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)

local player = Players:FindFirstChild("AXrain10")
if not player then
	player = Players.PlayerAdded:Wait()
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end
	task.wait(0.1)
end

if not data then
	warn("[TempGive] Data still not loaded for " .. tostring(player and player.Name))
	script:Destroy()
	return
end
-- 发植物
PlayerDataManager.GivePlant(player, PlantName, Amount, 1, true)
-- 发种子
--PlayerDataManager.GiveSeed(player, PlantName, Amount, true)

-- 打印信息
print(string.format("[TempGive] 成功发放！数量: %d, 植物: %s, 玩家: %s", Amount, PlantName, player.Name))

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```

```
local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)

	player = Players.PlayerAdded:Wait()

	data = PlayerDataManager.GetPlayerData(player)
```

```
	warn("[TempMaxLevel] Data still not loaded for " .. tostring(player and player.Name))

	return

local seasonId = RankingSeasonConfig.GetSeasonId()

if type(data.RankingSeasonUnlockedLevels) ~= "table" then
	data.RankingSeasonUnlockedLevels = {}
end

local oldSeasonLevel = tonumber(data.RankingSeasonUnlockedLevels[seasonId]) or 1
local oldUnlockedLevel = tonumber(data.UnlockedLevel) or 1

data.RankingSeasonUnlockedLevels[seasonId] = targetLevel
data.UnlockedLevel = targetLevel

if updateInventoryRE then
	updateInventoryRE:FireClient(player, data)
end

local character = player.Character
local head = character and character:FindFirstChild("Head")
local titleGui = head and head:FindFirstChild("PlayerTitleGui")
local titleLabel = titleGui and titleGui:FindFirstChildWhichIsA("TextLabel", true)
if titleLabel then
	titleLabel.Text = "MAX  " .. tostring(math.max(0, targetLevel - 1))
end

print(("[TempMaxLevel] %s season=%s max selectable level: %s -> %s"):format(
	player.Name,
	tostring(seasonId),
	tostring(oldSeasonLevel),
	tostring(targetLevel)
))
print(("[TempMaxLevel] UnlockedLevel: %s -> %s"):format(
	tostring(oldUnlockedLevel),
	tostring(data.UnlockedLevel)
))

if FORCE_SAVE then
	local ok = PlayerDataManager.ForceSave(player)
	print("[TempMaxLevel] ForceSave =", ok)
end

]]
```

## 5.发放药水

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempGiveMutationPotion")
if old then
	old:Destroy()
end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempGiveMutationPotion"
scriptObj.Source = [[
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerScriptService = game:GetService("ServerScriptService")

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)
local ItemConfig = require(ReplicatedStorage.Shared.ItemConfig)

-- 发放参数
local TARGET_PLAYER_NAME = "AXrain10"
local POTION_ITEM_ID = "ShinyPotion"
local AMOUNT = 1
local AUTO_EQUIP = false
local FORCE_SAVE = false

-- 可用药水：
-- ShinyPotion
-- ThunderPotion
-- DivinePotion

local function getItemQuantity(data, itemId)
	local inventory = data and data.Inventory
	local items = inventory and inventory.Items
	if type(items) ~= "table" then
		return 0
	end

	local quantity = 0
	for _, itemData in ipairs(items) do
		if itemData.ID == itemId then
			quantity += tonumber(itemData.Quantity) or 0
		end
	end

	return quantity
end

local function run()
	-- 验证数量，避免 0 或负数被底层接口修正为 1
	local amount = tonumber(AMOUNT)
	if not amount or amount <= 0 or amount % 1 ~= 0 then
		warn("[TempGivePotion] 发放失败：AMOUNT 必须是正整数")
		return
	end

	-- 验证药水配置
	local itemDetail = ItemConfig.Items and ItemConfig.Items[POTION_ITEM_ID]
	if not itemDetail or itemDetail.ItemType ~= "MutationPotion" then
		warn("[TempGivePotion] 发放失败：无效药水 ID：" .. tostring(POTION_ITEM_ID))
		return
	end

	-- 最多等待指定玩家 60 秒，不会误发给其他加入的玩家
	local player = Players:FindFirstChild(TARGET_PLAYER_NAME)
	for _ = 1, 600 do
		if player and player:IsA("Player") then
			break
		end

		task.wait(0.1)
		player = Players:FindFirstChild(TARGET_PLAYER_NAME)
	end

	if not player then
		warn("[TempGivePotion] 发放失败：未找到玩家 " .. TARGET_PLAYER_NAME)
		return
	end

	-- 等待玩家数据加载
	local data
	for _ = 1, 600 do
		if not player.Parent then
			break
		end

		data = PlayerDataManager.GetPlayerData(player)
		if data then
			break
		end

		task.wait(0.1)
	end

	if not data then
		warn("[TempGivePotion] 发放失败：玩家数据未加载 " .. player.Name)
		return
	end

	local beforeQuantity = getItemQuantity(data, POTION_ITEM_ID)

	local callOk, granted, compositeId = pcall(
		PlayerDataManager.GiveItem,
		player,
		POTION_ITEM_ID,
		amount,
		AUTO_EQUIP
	)

	if not callOk then
		warn("[TempGivePotion] GiveItem 运行错误：" .. tostring(granted))
		return
	end

	if granted ~= true then
		warn("[TempGivePotion] 发放失败：GiveItem 拒绝了本次发放")
		return
	end

	local afterData = PlayerDataManager.GetPlayerData(player)
	local afterQuantity = getItemQuantity(afterData, POTION_ITEM_ID)

	if afterQuantity ~= beforeQuantity + amount then
		warn(string.format(
			"[TempGivePotion] 药水已尝试发放，但数量核对异常：%d -> %d",
			beforeQuantity,
			afterQuantity
		))
		return
	end

	if FORCE_SAVE then
		local saved, saveError = PlayerDataManager.ForceSave(player)
		if not saved then
			warn(string.format(
				"[TempGivePotion] 药水已发到当前会话，但强制保存失败：%s",
				tostring(saveError)
			))
			return
		end
	end

	print(string.format(
		"[TempGivePotion] 发放成功！玩家: %s, 药水: %s, 数量: %d, 库存: %d -> %d, 自动装备: %s, 强制保存: %s",
		player.Name,
		POTION_ITEM_ID,
		amount,
		beforeQuantity,
		afterQuantity,
		tostring(AUTO_EQUIP),
		tostring(FORCE_SAVE)
	))
	print("[TempGivePotion] 背包 ID：" .. tostring(compositeId))
end

local ok, err = xpcall(run, debug.traceback)
if not ok then
	warn("[TempGivePotion] 未处理异常：\n" .. tostring(err))
end

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```

## 6.排行榜奖励测试

```
local ServerScriptService = game:GetService("ServerScriptService")

local old = ServerScriptService:FindFirstChild("TempTestRankingRewardFlow")
if old then old:Destroy() end

local scriptObj = Instance.new("Script")
scriptObj.Name = "TempTestRankingRewardFlow"
scriptObj.Source = [[
local Players = game:GetService("Players")
local DataStoreService = game:GetService("DataStoreService")
local ServerScriptService = game:GetService("ServerScriptService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local PlayerDataManager = require(ServerScriptService.Server.PlayerDataManager)
local GiftingService = require(ServerScriptService.Server.Services.GiftingService)
local RankingSeasonConfig = require(ReplicatedStorage.Shared.RankingSeasonConfig)

local TARGET_PLAYER_NAME = "AXrain10"
local TEST_RANK = 1 -- 改成 1/2/5/6/10/11/20/21/60/61 测边界
local TEST_SCORE = 1500

local player = Players:FindFirstChild(TARGET_PLAYER_NAME) or Players:GetPlayers()[1]
if not player then
	warn("[RankRewardTest] No player found. Start Play first.")
	script:Destroy()
	return
end

local data
for i = 1, 600 do
	data = PlayerDataManager.GetPlayerData(player)
	if data then break end
	task.wait(0.1)
end
if not data then
	warn("[RankRewardTest] Player data not loaded: " .. player.Name)
	script:Destroy()
	return
end

local testStoreName = "TEST_RankingRewardFlow_" .. tostring(player.UserId) .. "_" .. tostring(os.time())
local testSettlementId = "TEST_" .. tostring(RankingSeasonConfig.SettlementId) .. "_" .. tostring(TEST_RANK) .. "_" .. tostring(os.time())
local store = DataStoreService:GetOrderedDataStore(testStoreName)

for i = 1, math.max(0, TEST_RANK - 1) do
	store:SetAsync(tostring(900000000 + i), TEST_SCORE + i)
end
store:SetAsync(tostring(player.UserId), TEST_SCORE)

local pages = store:GetSortedAsync(false, RankingSeasonConfig.TopCount or 60)
local actualRank
for rank, entry in ipairs(pages:GetCurrentPage()) do
	if tostring(entry.key) == tostring(player.UserId) then
		actualRank = rank
		break
	end
end

print("[RankRewardTest] target =", player.Name, "expected rank =", TEST_RANK, "actual rank =", tostring(actualRank))

if not actualRank then
	print("[RankRewardTest] Target not in top list, no reward queued. This is expected for rank > TopCount.")
	script:Destroy()
	return
end

local tier = RankingSeasonConfig.GetRewardForRank(actualRank)
if not tier or type(tier.Rewards) ~= "table" then
	print("[RankRewardTest] No reward tier for rank", actualRank)
	script:Destroy()
	return
end

local gifts = {}
for index, reward in ipairs(tier.Rewards) do
	local rewardType = reward.RewardType or reward.Type
	local itemId = reward.ItemId or reward.PlantId or reward.SeedId or reward.ChestId or reward.TitleId
	local amount = math.max(1, math.floor(tonumber(reward.Amount) or 1))
	local displayName = reward.DisplayName or itemId or rewardType
	table.insert(gifts, {
		Id = testSettlementId .. "_" .. tostring(player.UserId) .. "_" .. tostring(index) .. "_" .. tostring(rewardType) .. "_" .. tostring(itemId),
		Title = "TEST Leaderboard Season Reward",
		RewardType = rewardType,
		ItemId = itemId,
		Amount = amount,
		Mutation = reward.Mutation or reward.MutationLevel,
		RewardText = ("TEST Rank #%d: %dx %s"):format(actualRank, amount, tostring(displayName)),
		CreatedAt = os.time(),
	})
	print("[RankRewardTest] queue", rewardType, itemId, amount)
end

local ok, message = GiftingService.QueueGiftsForUserId(player.UserId, gifts)
print("[RankRewardTest] Queue result =", ok, message)

task.wait(3)
data = PlayerDataManager.GetPlayerData(player)
print("[RankRewardTest] Pending gifts:")
for _, gift in ipairs(data.PendingGifts or {}) do
	if string.find(gift.Id, testSettlementId, 1, true) then
		print(" ", gift.Id, gift.RewardText)
	end
end

script:Destroy()
]]
scriptObj.Parent = ServerScriptService
```
